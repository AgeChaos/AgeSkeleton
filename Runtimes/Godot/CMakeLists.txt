cmake_minimum_required(VERSION 3.20)
project(AgeSkeletonGodot LANGUAGES CXX)
set(GODOT_CPP_PATH "" CACHE PATH "Path to official godot-cpp 4.4 sources")
if(NOT EXISTS "${GODOT_CPP_PATH}/CMakeLists.txt")
  message(FATAL_ERROR "Set GODOT_CPP_PATH to a godot-cpp 4.4 checkout")
endif()
add_subdirectory("${GODOT_CPP_PATH}" godot-cpp)
add_library(ageskeleton_godot SHARED src/age_skeleton_player.cpp)
target_include_directories(ageskeleton_godot PRIVATE ../Native/include)
target_link_libraries(ageskeleton_godot PRIVATE godot-cpp::template_release)
target_compile_features(ageskeleton_godot PRIVATE cxx_std_17)
set_target_properties(ageskeleton_godot PROPERTIES PREFIX "" OUTPUT_NAME "ageskeleton_godot")
